// =============================================================================
// FIRESTORE SECURITY RULES — v3 DA PUBBLICARE (audit monitoraggio 2026-10-27)
// =============================================================================
// Costruita sulle regole REALMENTE pubblicate (backup in
// firestore.rules.current.txt, scaricato dall'API il 27/10/2026).
//
// Cosa cambia rispetto alla v2 in produzione:
//   1. 'app': un utente attivo che NON è staff (cioè un AGENTE) può ancora
//      leggere e aggiornare il doc (Report App salva le statistiche), ma NON può
//      toccare i campi del monitoraggio (monitorFields: User ID fantasma, slug,
//      Webzine ID, toggle, stato sync), né creare app col monitor né eliminarle.
//   2. 'push_history': scrittura solo staff (prima: qualunque utente attivo).
//   3. 'health_status': lettura solo staff, scrittura solo server (prima era
//      nella whitelist generica: un agente poteva riscriverla).
// Staff = SUPER_ADMIN, ADMIN, CTO, SVILUPPATORE, CONTENT_MANAGER, CONTABILE
// (gli stessi ruoli che aprono Storico Push; stessa lista di lib/staff-auth.js).
//
// Collaudo 27/10/2026 con l'API firebaserules ":test" (nessuna pubblicazione):
// 21/21 casi OK (agente bloccato su User ID, toggle, slug, creazione, delete,
// push_history, health_status; staff e pagine pubbliche invariati). Gli stessi
// casi sulla v2 in produzione: 9 falle.
//
// IMPORTANTE: in Firestore Rules tutti i match si valutano in OR (union).
// NON sostituire la whitelist isCrmCollection con un /{document=**} permissivo.
// ⚠️ APERTO: 'push_history' resta leggibile da chiunque con campi interni
//    (gbUrl, sentBy...). Serve uno specchio 'push_public' (vedi audit 2026-09-18).
// =============================================================================

rules_version = '2';

// FIRESTORE SECURITY RULES — FASE A (v3: monitoraggio notifiche solo staff)
// v3 (audit 27/10/2026): gli AGENTI non possono più toccare il monitoraggio
// notifiche (User ID fantasma, slug, toggle, stato sync) né lo storico push;
// health_status lo scrive solo il server e lo legge solo lo staff.
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuth() {
      return request.auth != null;
    }

    function userDoc() {
      return get(/databases/$(database)/documents/utenti/$(request.auth.uid)).data;
    }

    function isActive() {
      return isAuth()
        && exists(/databases/$(database)/documents/utenti/$(request.auth.uid))
        && userDoc().get('stato', 'ATTIVO') != 'DISATTIVO';
    }

    function userRole() {
      return userDoc().ruolo;
    }

    function isSuperAdmin() {
      return isActive() && userRole() == 'SUPER_ADMIN';
    }
    function isAdmin() {
      return isActive() && userRole() in ['SUPER_ADMIN', 'ADMIN'];
    }

    // Staff = chi può aprire Storico Push nel CRM (view_all_data o manage_apps
    // in public/js/auth.js). NON l'AGENTE. Stessa lista di lib/staff-auth.js.
    function isStaff() {
      return isActive() && userDoc().get('ruolo', '') in
        ['SUPER_ADMIN', 'ADMIN', 'CTO', 'SVILUPPATORE', 'CONTENT_MANAGER', 'CONTABILE'];
    }

    // Campi del doc 'app' che governano il monitoraggio notifiche.
    function monitorFields() {
      return ['pushMonitorEnabled', 'monitorPushUserId', 'appSlug', 'goodbarberWebzineId',
              'lastPushSync', 'lastPushSyncStatus', 'lastPushSyncError', 'lastPushSyncWarning',
              'lastPushSyncCount', 'lastPushSyncSkipped', 'lastPushSyncFuture',
              'lastPushSyncPagesScanned', 'lastPushSyncErrorStreak', 'lastPushSyncOkAt'];
    }

    function isOwnDoc(userId) {
      return isAuth() && request.auth.uid == userId;
    }

    function onlyAffects(allowedFields) {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(allowedFields);
    }

    // Whitelist delle 29 collezioni CRM "standard"
    function isCrmCollection(coll) {
      return coll in [
        'app_followers', 'chat_conversations', 'clienti', 'commenti', 'contatori',
        'contratti', 'discussioni_app', 'documenti', 'fatture', 'generatore-home-configs',
        'impostazioni', 'messages', 'note_cliente', 'notifications',
        'od_attivita', 'od_istanze_componenti', 'od_prodotti', 'presence', 'promemoria',
        'push_broadcast_log', 'saleRiunioni', 'scadenzario', 'settings', 'tasks',
        'rss_feeds', 'rss_assignments', 'rss_classifications', 'rss_read',
        'od_componenti', 'od_changelog',
        'fornitori', 'fornitori_valutazioni'
      ];
    }


    // COLLEZIONI PUBBLICHE

    // Specchio pubblico minimale (solo nome/slug), scritto solo dal server (Admin SDK)
    match /app_public/{document} {
      allow read: if true;
      allow write: if false;
    }
    // 'app': tutti gli utenti attivi leggono e aggiornano (es. Report App degli
    // agenti salva le statistiche), ma i campi del monitoraggio li tocca solo lo staff.
    match /app/{document} {
      allow read: if isActive();
      allow create: if isStaff()
        || (isActive() && !request.resource.data.keys().hasAny(monitorFields()));
      allow update: if isStaff()
        || (isActive() && !request.resource.data.diff(resource.data).affectedKeys().hasAny(monitorFields()));
      allow delete: if isStaff();
    }
    // Storico notifiche: letto dalle pagine pubbliche, scritto dal server e dallo
    // staff (Push Broadcast registra gli invii). Gli agenti non scrivono.
    match /push_history/{document} {
      allow read: if true;
      allow write: if isStaff();
    }
    // Fotografia giornaliera della salute del monitoraggio: la scrive solo il
    // cron daily-health-alert (Admin SDK), la legge solo lo staff (banner Dashboard).
    match /health_status/{document} {
      allow read: if isStaff();
      allow write: if false;
    }


    // /utenti — anti-escalation
    match /utenti/{userId} {
      allow read: if isOwnDoc(userId) || isActive();
      allow create: if isAdmin();
      allow update: if isAdmin()
        || (isOwnDoc(userId) && isActive() && onlyAffects([
              'lastLogin', 'photoURL', 'dataAggiornamento',
              'fcmTokens', 'fcmTokenUpdatedAt'
           ]));
      allow delete: if isSuperAdmin();
    }


    // /audit_log — append-only
    match /audit_log/{document} {
      allow read: if isAdmin();
      allow create: if isActive();
      allow update, delete: if false;
    }


    // Altre collezioni CRM (whitelist)
    match /{collection}/{document} {
      allow read, write: if isCrmCollection(collection) && isActive();
    }

    // Catch-all: tutto il resto e' bloccato
    // Messaggi chat: subcollection non coperta dal match a segmento singolo (v10.2.0)
    match /chat_conversations/{convId}/messages/{msgId} {
      allow read, write: if isActive();
    }

    match /{path=**} {
      allow read, write: if false;
    }
  }
}
